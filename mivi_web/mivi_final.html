python3 -c "import http.server,urllib.parse,urllib.request as u;
class H(http.server.BaseHTTPRequestHandler):
 def do_GET(s):s.r()
 def do_POST(s):s.r()
 def r(s):
  q=urllib.parse.urlparse(s.path);t=urllib.parse.parse_qs(q.query).get('u',[''])[0]
  if t:
   b=s.rfile.read(int(s.headers.get('Content-Length') or 0)) or None
   rq=u.Request(t,data=b,method=s.command)
   for k,v in s.headers.items():
    if k.lower() not in ('host','origin','referer','content-length'):rq.add_header(k,v)
   try:
    with u.urlopen(rq) as p:body=p.read();code=p.status
   except Exception as e:body=str(e).encode();code=502
   s.send_response(code)
   for k,v in (('Access-Control-Allow-Origin','*'),('Content-Type','application/json')):s.send_header(k,v)
   s.end_headers();s.wfile.write(body)
  else:s.send_response(200);s.end_headers();s.wfile.write(b'proxy ok')
 def log_message(s,*a):pass
http.server.ThreadingHTTPServer(('127.0.0.1',8770),H).serve_forever()"
